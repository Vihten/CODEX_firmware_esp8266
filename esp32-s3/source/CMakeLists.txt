cmake_minimum_required(VERSION 3.16)
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
idf_build_set_property(MINIMAL_BUILD ON)
project(led_forge_s3)

# Backport Espressif 0d8ff68f8a47fc05a6b1d3a5bda24764a48e5fbd for IDF 6.0.1.
# Keep the SDK installation untouched and replace only the bundle translation unit.
idf_component_get_property(codex_mbedtls_lib mbedtls COMPONENT_LIB)
get_target_property(codex_mbedtls_sources ${codex_mbedtls_lib} SOURCES)
set(codex_bundle_source_found FALSE)
set(codex_patched_sources "")
foreach(codex_source IN LISTS codex_mbedtls_sources)
    if(codex_source MATCHES "(^|/)esp_crt_bundle\\.c$")
        list(APPEND codex_patched_sources "${CMAKE_CURRENT_SOURCE_DIR}/patches/esp_crt_bundle.c")
        set(codex_bundle_source_found TRUE)
    else()
        list(APPEND codex_patched_sources "${codex_source}")
    endif()
endforeach()
if(NOT codex_bundle_source_found)
    message(FATAL_ERROR "Expected ESP-IDF certificate bundle source was not found")
endif()
set_property(TARGET ${codex_mbedtls_lib} PROPERTY SOURCES "${codex_patched_sources}")
