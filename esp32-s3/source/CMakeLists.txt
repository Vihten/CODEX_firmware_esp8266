cmake_minimum_required(VERSION 3.16)
set(PROJECT_VER "1795172000000")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
idf_build_set_property(MINIMAL_BUILD ON)
project(led_forge_s3)

# Backport Espressif 0d8ff68f8a47fc05a6b1d3a5bda24764a48e5fbd for IDF 6.0.1.
# Keep the SDK installation untouched and replace only the bundle translation unit.
idf_component_get_property(codex_mbedtls_lib mbedtls COMPONENT_LIB)
get_target_property(codex_mbedtls_sources ${codex_mbedtls_lib} SOURCES)
set(codex_bundle_source_found FALSE)
set(codex_patched_sources "")
foreach(codex_source IN LISTS codex_mbedtls_sources)
    if(codex_source MATCHES "(^|/)esp_crt_bundle\\.c$")
        list(APPEND codex_patched_sources "${CMAKE_CURRENT_SOURCE_DIR}/patches/esp_crt_bundle.c")
        set(codex_bundle_source_found TRUE)
    else()
        list(APPEND codex_patched_sources "${codex_source}")
    endif()
endforeach()
if(NOT codex_bundle_source_found)
    message(FATAL_ERROR "Expected ESP-IDF certificate bundle source was not found")
endif()
set_property(TARGET ${codex_mbedtls_lib} PROPERTY SOURCES "${codex_patched_sources}")

# Capture the exact overdue tasks while the SDK still owns the watchdog lock.
# Generate a patched translation unit in the build directory, leaving IDF intact.
set(codex_wdt_original "$ENV{IDF_PATH}/components/esp_system/task_wdt/task_wdt.c")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${codex_wdt_original}")
file(READ "${codex_wdt_original}" codex_wdt_code)
set(codex_wdt_anchor "    if (esp_task_wdt_print_triggered_tasks(NULL, NULL, &cpus_fail) != ESP_OK) {")
string(FIND "${codex_wdt_code}" "${codex_wdt_anchor}" codex_wdt_position)
if(codex_wdt_position EQUAL -1)
    message(FATAL_ERROR "Expected ESP-IDF watchdog ISR capture point was not found")
endif()
string(REPLACE "${codex_wdt_anchor}" "    extern void codex_capture_task_watchdog(void);\n    codex_capture_task_watchdog();\n${codex_wdt_anchor}" codex_wdt_code "${codex_wdt_code}")
set(codex_wdt_patched "${CMAKE_CURRENT_BINARY_DIR}/codex_task_wdt.c")
file(CONFIGURE OUTPUT "${codex_wdt_patched}" CONTENT "${codex_wdt_code}" @ONLY)
idf_component_get_property(codex_system_lib esp_system COMPONENT_LIB)
get_target_property(codex_system_sources ${codex_system_lib} SOURCES)
list(FILTER codex_system_sources EXCLUDE REGEX "(^|/)task_wdt/task_wdt\\.c$")
get_target_property(codex_system_original_sources ${codex_system_lib} SOURCES)
if(codex_system_sources STREQUAL codex_system_original_sources)
    message(FATAL_ERROR "Expected ESP-IDF task watchdog source was not found")
endif()
list(APPEND codex_system_sources "${codex_wdt_patched}")
set_property(TARGET ${codex_system_lib} PROPERTY SOURCES "${codex_system_sources}")
